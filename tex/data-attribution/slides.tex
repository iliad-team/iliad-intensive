\begin{frame}{A science of generalisation}
  \textbf{Why} do models generalise? \textbf{How?}

  We want a \emph{science} of this map.
  \vfill
  \begin{itemize}
    \item \textbf{Mechanistic interpretability} is causal analysis of the \emph{forward pass}: which circuits caused this output?
    \item \textbf{Data attribution} is causal analysis of the \emph{training run}: which data caused this model?
  \end{itemize}

\end{frame}

% -------------------------------------------------------------
\begin{frame}{Why attribute?}

  \begin{itemize}
    \item \textbf{Debugging \& Curation.} Identify unwanted behaviour's causes and filter them out. Amend aligned behaviour's causes to make it more aligned.
    \item \textbf{Memorisation \& Generalization.} Did the model strictly pattern match or make inference beyond the training data? How does structure in data become structure in models?
  \end{itemize}
\end{frame}

\begin{frame}{Distributional vs.\ single-model attribution}
  Which \emph{model} are we attributing to?
  \vfill
  \begin{itemize}
    \item \textbf{Single-model.} Understand the causes of a \emph{particular} model's behaviour. What data caused this model to respond in this way?
    \item \textbf{Distributional.} Understand the causes of a \emph{distribution} of models' behaviour. What data causes models in general to respond in this way
  \end{itemize}

\end{frame}

% -------------------------------------------------------------
\begin{frame}{Entangled latent concepts}
  Attribution scores a \emph{behaviour} --- but behaviours bundle many causes.
  \vfill
  \begin{center}
    \emph{Q: What is the capital of France? \quad A: Paris.}
  \end{center}
  \vfill
  What caused the correct answer?
  \begin{itemize}
    \item \textbf{Factual knowledge} --- the Wikipedia article on Paris.
    \item \textbf{Language understanding} --- grammar, Q\&A format, tokenisation.
    \item \textbf{Format priors} --- capital-city questions in pretraining.
  \end{itemize}
  \vfill
\end{frame}
