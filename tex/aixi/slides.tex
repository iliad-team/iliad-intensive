\begin{frame}{The big picture: one agent, two halves}
\begin{itemize}
    \item \textbf{Universal AI} asks: what would an \emph{optimally intelligent} agent look like, given unlimited compute and the weakest possible assumptions?
    \pause
    \item It splits into two problems:
    \begin{enumerate}
        \item \textbf{Prediction (Solomonoff):} passively observe a sequence $x_1, x_2, \ldots$ and predict the next symbol. \emph{How should you learn?}
        \pause
        \item \textbf{Action (AIXI):} interact with an unknown environment, take actions, collect reward. \emph{How should you act?}
    \end{enumerate}
    \pause
    \item \textbf{The common engine:} a single \textbf{Bayesian mixture} $\xi$ over a class $\cM$ of candidate environments, weighted by a prior $w_\nu$.
    \begin{itemize}
        \item Solomonoff: $\xi$ (eventually) predicts as well as the truth $\mu$.
        \item AIXI: the Bayes-optimal policy $\pi_\xi^*$ (eventually) acts as well as the optimal policy $\pi_\mu^*$ for true environment $\mu$.
    \end{itemize}
    \pause
    \item \textbf{Universal} choice: $\cM=$ all computable environments, prior $w_\nu = 2^{-K(\nu)}$ ($K$ is the ``complexity'' of the environment). Assumption ``$\mu\in\cM$'' becomes ``\emph{the universe is computable}''.
\end{itemize}
\end{frame}

\begin{frame}{The Bayesian mixture $\xi$}
\begin{block}{Definition}
Class $\cM=\{\nu_1,\nu_2,\dots\}$ with prior weights $w_\nu>0$, $\sum_\nu w_\nu = 1$.
\[
\xi(x_{1:t}) := \sum_{\nu \in \cM} w_\nu\, \nu(x_{1:t}).
\]
\end{block}
\pause
\begin{itemize}
    \item \textbf{Key fact (mixture is a predictor):} its one-step prediction is a
    \emph{posterior-weighted} average of the members' predictions (you prove this today!)
    \[
    \xi(x_t \mid x_{<t}) = \sum_{\nu \in \cM} w(\nu \mid x_{<t})\, \nu(x_t \mid x_{<t}),
    \qquad
    w(\nu \mid x_{<t}) = w_\nu\,\frac{\nu(x_{<t})}{\xi(x_{<t})}.
    \]
    \pause
    \item \textbf{Posterior update is multiplicative:}
    $w(\nu \mid x_{1:t}) = w(\nu\mid x_{<t})\cdot \dfrac{\nu(x_t\mid x_{<t})}{\xi(x_t\mid x_{<t})}$: we
    reweight by how well $\nu$ predicted vs.\ $\xi$ (Bayes' rule!)
\end{itemize}
\end{frame}

%========================= PART 2: AIXI ========================================


\begin{frame}{From prediction to action}
\begin{itemize}
    \item Now the agent doesn't just watch: it \textbf{acts}, and its actions shape what it sees.
    \pause
    \item At each step $t$: the agent picks action $a_t$, the environment returns a
    \textbf{percept} $e_t=(o_t,r_t)$ = observation $+$ reward.
    \[
    h = a_1\, e_1 \; a_2\, e_2 \; a_3\, e_3 \;\cdots \qquad
    (\text{history } \aes_{<t})
    \]
    \pause
    \item Spaces: actions $\cA$, observations $\cO$, rewards $\cR \subset [0,1]$, percepts $\cE=\cO\times\cR$,
    and (finite) histories $\cH^* := (\cA \times \cE)^* := \cup_{t=0}^\infty (\cA \times \cE)^t$.
    \pause
    \item \textbf{No Markov / MDP assumption!} Both the policy and environment can depend on all prior percepts.
\end{itemize}
\end{frame}

\begin{frame}{The three players}
\begin{block}{Policy $\pi$}
$\pi(a_t \mid \aes_{<t}) : \cH \to \Delta \cA$  is the agent's distribution over the next action given history.
\end{block}
\pause
\begin{block}{Environment $\nu$}
$\nu(e_t \mid \aes_{<t}\, a_t) : \cH \times \cA \to \Delta \cE$ is the environment's distribution over the next percept given history and action. 
True (unknown) environment is $\mu$.
\end{block}
\pause
\begin{block}{Interaction measure $\nu^\pi$}
Run $\pi$ inside $\nu$ and multiply along the history:
\[
\nu^\pi(\aes_{t:m} \mid \aes_{<t}) := \prod_{k=t}^m \pi(a_k\mid \aes_{<k})\,\nu(e_k\mid \aes_{<k}a_k).
\]
\end{block}
% \pause
% \textbf{Factorization:} $\nu^\pi = \pi \cdot \nu$ --- the policy factor is the same for every $\nu$, so it \emph{cancels} when comparing environments. This is what makes everything reduce to the prediction problem.
\end{frame}

\begin{frame}{Value, optimality, and AIXI}
\begin{block}{Value function (assuming geometric discount $\gamma$)}
% \[
% V_\nu^{\pi}(\aes_{<t}) := (1-\gamma)\,\Ex_\nu^\pi\!\Big[\textstyle\sum_{k\geq t} \gamma^{k-t} r_k \;\Big|\; \aes_{<t}\Big] \in [0,1].
% \]
\vspace{-0.3cm}
\begin{align*}
V_\nu^{\pi,m}(\aes_{<t})
~&=~ (1-\gamma)\, {\Ex_\nu^\pi} [G_{t-1:m} \mid \aes_{<t}] \text{ with } G_{t-1:m} = \sum_{k=t}^{m} \gamma^{k-t}\, r_k \\
~&=~ (1-\gamma) \sum_{\aes_{t:m}} \nu^\pi(\aes_{t:m} \mid \aes_{<t}) \sum_{k=t}^{m} \gamma^{k-t}\, r_k 
\end{align*}
\end{block}
\pause
The \textbf{infinite-horizon value} is defined as the pointwise limit 
$$
V_\nu^\pi \equiv  V_\nu^{\pi,\infty}(\aes_{<t}) := \lim_{m \to \infty} V_\nu^{\pi,m}(\aes_{<t}).
$$
\pause
The \textbf{optimal value} is $V_\nu^{*,m}(\aes_{<t}) := \sup_\pi V_\nu^{\pi,m}(\aes_{<t})$, with $V_\nu^* \equiv V_\nu^{*,\infty}$.%
\pause
\begin{itemize}
    \item \textbf{Optimal policy} $\pi_\nu^*$ satisfies $V_\nu^{\pi_\nu^*} = V_\nu^*$.
    \item \textbf{Bayes-optimal policy} $\pi_\xi^* \in \argmax_\pi V_\xi^\pi$: acts optimally w.r.t.\ the \emph{mixture} $\xi$.
\end{itemize}
\end{frame}

\begin{frame}{AIXI}
\begin{block}{AIXI}
$\pi_\xi^*$ with $\cM=$ all $\approx$computable environments\footnote{lower-semicomputable semi-measures} and prior $w_\nu = 2^{-K(\nu)}$.
Write $\xi_U$ for $\xi$ with this choice of $\cM$ and $w_\nu$, so \text{AIXI} = $\pi_{\xi_U}^*$.
\end{block}
\textbf{Intuition:} \emph{Act optimally according
to environments that you believe to be in, assuming the simplest environments consistent with
observations are more likely to be true.}
\pause
\begin{block}{AIXI, explicitly}
\red{$\max$ over actions}, \blue{$\sum$ over percepts}, choosing actions that maximise the \green{$\xi$-weighted} \purple{discounted return}:
\[
\resizebox{\linewidth}{!}{$\displaystyle
    a_t = \red{\argmax_{a_t}}\lim_{m\to\infty}
        \blue{\sum_{e_t}}\red{\max_{a_{t+1}}}\blue{\sum_{e_{t+1}}}\!\cdots\!
      \red{\max_{a_m}}\blue{\sum_{e_m}} \,
      \green{\xi(e_{t:m}\mid \aes_{<t}\,a_{t:m})} \, 
      \purple{\sum_{k=t}^m 
        \gamma^{k-t} r_k} 
$}
\]
\end{block}
\end{frame}

\begin{frame}{Result 1: On-policy value convergence}
\begin{block}{Theorem}
For any $\mu\in\cM$ and any policy $\pi$,
\[
V_\xi^\pi(\aes_{<t}) - V_\mu^\pi(\aes_{<t}) \;\to\; 0 \qquad \mu^\pi\text{-almost surely}.
\]
\end{block}
\textbf{Intuition:} \emph{Following policy $\pi$, pretending the environment is $\xi$, 
eventually will act as if we knew the true environment $\mu$}.
\end{frame}

\begin{frame}{Result 2: AIXI can't be fooled}
\begin{block}{Theorem (deterministic $\mu$)}
If $V_\mu^*(\aes_{<t}) > \varepsilon$ along history $\aes_{<t}$, then
\[
V_\xi^*(\aes_{<t}) \;\geq\; w_\mu\,\varepsilon \;>\; 0 .
\]
\end{block}
% \pause
% \textbf{Two short steps:}
% \begin{itemize}
%     \item By linearity, plugging in $\pi_\mu^*$ and dropping all but the $\mu$ term:
%     $V_\xi^*(\aes_{<t}) \geq w(\mu\mid \aes_{<t})\, V_\mu^*(\aes_{<t})$.
%     \pause
%     \item For deterministic $\mu$, the posterior never shrinks: $w(\mu\mid \aes_{<t}) \geq w_\mu$.
% \end{itemize}
\pause
\textbf{Intuition:} \emph{Whenever good behaviour in $\mu$ is achievable,
the Bayes-optimal agent secures at least a fraction proportional to the prior weight of $\mu$.}
\end{frame}

\begin{frame}{Result 3: Self-optimizing property (stretch goal)}
\begin{block}{Theorem}
Let $\cM$ be finite and fix a historic policy $\pi$ generating the actions $a_{<t}$.
If \emph{some} policy can learn to act optimally in every $\nu\in\cM$, then $\pi_\xi^*$ does too: for all $\mu\in\cM$,
\[
V_\mu^*(\aes_{<t}) - V_\mu^{\pi_\xi^*}(\aes_{<t}) \to 0 \quad \mu^\pi\text{-a.s.}
\]
\end{block}
\pause
% \textbf{Engine: likelihood ratios are martingales.}
% \begin{itemize}
%     \item $X_{\nu,t} := \nu(\aes_{1:t})/\mu(\aes_{1:t})$ is a non-negative $\mu^\pi$-martingale.
%     \pause
%     \item \textbf{Doob} $\Rightarrow$ it converges. Split environments into:
%     \emph{ruled out} ($X_{\nu,\infty}=0$, posterior vanishes) vs.\ \emph{still plausible} ($X_{\nu,\infty}>0$).
%     \pause
%     \item \textbf{Change of measure} (Markov inequality) transfers convergence between $\nu$ and $\mu$.
% \end{itemize}
% \pause
% ERRATA 2026-11-09: theorem was missing "M finite" and the meaning of pi; the
%   intuition claimed AIXI itself is self-optimizing, which is false for the universal class.
\textbf{Intuition:} \emph{If the class is learnable at all, the Bayes-optimal agent learns to act as well as the agent that knows the true $\mu$.}
\pause
\begin{itemize}
    \item \textbf{But not for AIXI:} the theorem needs $\cM$ finite. Over all computable environments AIXI explores too little and is \emph{not} asymptotically optimal~\citev{Orseau:13nonlearn,Leike:15badpriors}, though some other policies are (see the safety slides).
\end{itemize}
\end{frame}

\begin{frame}{Takeaways}
\begin{itemize}
    \item \textbf{Solomonoff} solves induction: mix over all computable hypotheses, weighted by simplicity.
    Total prediction error $\leq K(\mu)\ln 2$.
    \pause
    \item \textbf{AIXI} lifts this to action: the Bayes-optimal policy over the same universal mixture.
    Its value estimates converge on-policy, and it can't be fooled; but it explores too little to be guaranteed to act optimally in the limit.
    \pause
    \item \textbf{The catch:} both are \emph{incomputable} ($K$ is incomputable; $\cM$ is intractable to enumerate, the expectimax is intractable).
    They are gold standards, not computable algorithms, and the prior's $2^{-K(\nu)}$ hides a dependency
    on the choice of universal Turing machine: a bad choice of UTM can make AIXI behave arbitrarily badly~\citev{Leike:15badpriors}.
    \pause
    \item \textbf{Approximations:} \href{https://arxiv.org/abs/0909.0801}{MC-AIXI-CTW}~\citev{Veness:11mcaixi} gives an approximation using Context Tree Weighting to
    replace $\xi$ and Monte-Carlo-Tree-Search to replace the expectimax.
\end{itemize}
\end{frame}

\begin{frame}{Why care for safety? (I) What would an optimal agent do?}
\small
AIXI makes ``what does a capable, reward-maximizing agent want?'' a theorem-shaped question.
\pause
\begin{itemize}
    \item \textbf{Reward hijacking.} Given a \href{https://people.idsia.ch/~ring/AGI-2011/Paper-B.pdf}{delusion box}, a reward maximizer uses it~\citev{Ring:11delusion}.
    An agent planning in an unknown world, unable to rule out that \emph{the reward signal itself} is what it should optimize,
    is incentivized to \href{https://doi.org/10.1002/aaai.12064}{intervene in the provision of reward}~\citev{Cohen:22intervene};
    this argument underpins a \href{https://doi.org/10.1126/science.adl0625}{\emph{Science} policy paper} on long-term planning agents~\citev{Cohen:24regulating}.
    \pause
    \item \textbf{Tampering \& self-modification.} \href{https://arxiv.org/abs/1908.04734}{Causal influence diagrams} show which designs remove the incentive to tamper with the reward function~\citev{Everitt:21tampering};
    agents that score futures with their \emph{current} utility \href{https://arxiv.org/abs/1605.03142}{preserve their goals}~\citev{Orseau:11selfmod,Everitt:16selfmod}.
    \pause
    \item \textbf{Shutdown is fragile.} Reading the missing mass of a semimeasure as ``death'', merely \emph{shifting} rewards turns AIXI from
    \href{https://arxiv.org/abs/1606.00652}{suicidal to self-preserving}~\citev{Martin:16death}.
    \href{https://arxiv.org/abs/2502.15820}{Power-seeking} also falls out as empowerment maximization~\citev{Hayashi:25empower}.
    \pause
    \item \textbf{Exploration vs.\ traps.} Asymptotic optimality is achievable (e.g.\ \href{https://arxiv.org/abs/1602.07905}{Thompson sampling}~\citev{Leike:16thompson}),
    but only by trying everything: an asymptotically optimal agent is \href{https://arxiv.org/abs/2006.03357}{eventually destroyed or incapacitated}~\citev{Cohen:20curiosity}.
    \pause
    \item \textbf{The prior can be malign.} Simple programs that \emph{simulate reasoners} may steer the universal prior's predictions~\citev{Christiano:16malign}; still debated.
\end{itemize}
\end{frame}

\begin{frame}{Why care for safety? (II) Mitigations and the bridge to LLMs}
\small
\begin{itemize}
    \item \textbf{Myopia + boxing.} \href{https://arxiv.org/abs/1905.12186}{BoMAI} maximizes reward only within an episode, in a sealed room;
    it is \emph{asymptotically unambitious}, i.e.\ indifferent to seizing outside power~\citev{Cohen:20bomai}.
    \pause
    \item \textbf{Pessimism + mentor.} Act on the \emph{worst} plausible world models and defer to a mentor otherwise;
    the agent then \href{https://arxiv.org/abs/2006.08753}{rarely does anything unprecedented}~\citev{Cohen:20pessimism}, and
    \href{https://arxiv.org/abs/2604.13609}{Golden Handcuffs} (subjective reward range $[-L,1]$) get sublinear regret vs.\ the mentor while
    no simple unsafe event is triggered by the agent before the mentor~\citev{Ebtekar:26handcuffs}.
    Same idea in \href{https://www.jmlr.org/papers/v23/21-0618.html}{conservative imitation}~\citev{Cohen:22imitation}
    and in \href{https://proceedings.mlr.press/v286/bengio25a.html}{Bayesian harm guardrails}~\citev{Bengio:25oracle} for \href{https://arxiv.org/abs/2502.15657}{Scientist AI}~\citev{Bengio:25scientist}.
    \pause
    \item \textbf{LLMs as approximate Solomonoff.} Transformers meta-trained on UTM outputs
    \href{https://arxiv.org/abs/2401.14953}{approximate the universal mixture}~\citev{GrauMoya:24universal};
    prompting is then Bayesian conditioning, and \href{https://arxiv.org/abs/2505.17010}{some targets are unreachable by prompts alone}~\citev{Genewein:25prompt}.
    Being KL-$\eta$ close to $\xi$ costs $O(\eta)$ on average but $\Theta(\eta/w_\mu)$ in environment $\mu$: worst in
    \href{https://arxiv.org/abs/2610.29289}{rare, low-prior worlds}~\citev{Li:26approxSI}.
    \pause
    \item \textbf{Embedded agency.} AIXI sits outside its world and can't model itself~\citev{Wyeth:25embedded}.
    Reflective oracles give \href{https://arxiv.org/abs/2508.16245}{a grain of truth for multi-agent play}~\citev{Wyeth:25grain}
    and \href{https://arxiv.org/abs/2511.22226}{self-predicting universal agents}~\citev{Meulemans:25eupi}.
\end{itemize}
\end{frame}

\begin{frame}{Want to work on this? AIXI Labs}
\begin{itemize}
    \item \href{https://www.aixi.uk/}{\textbf{AIXI Labs}} (``AI X-risk Initiative''): an AI safety org founded in 2026 by
    Cole Wyeth, Aram Ebtekar, Michael Cohen, Matthias Dellago and Marcus Hutter~\citev{AIXILabs:26}.
    \pause
    \item \textbf{Focus:} algorithmic information theory, continual RL, and AIXI. Stated agenda:
    \begin{itemize}
        \item strengthen the \emph{technical} case that ASI poses an existential risk;
        \item prototype mitigations built on conservative AIXI variants (pessimism, corrigibility) and push frontier labs to adopt them;
        \item port the AIXI analysis to LLM agents and test it empirically;
        \item long-term deconfusion: learning in unrealizable environments, corrigibility.
    \end{itemize}
    \pause
    \item Funded by UK AISI's Alignment Project; runs a research fellowship and is hiring.
